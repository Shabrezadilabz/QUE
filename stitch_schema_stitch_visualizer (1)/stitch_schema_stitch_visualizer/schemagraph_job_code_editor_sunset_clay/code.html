import polars as pl
from schemagraph.core import transform

# Initialize transformation context
@transform(source="s3://raw-events")
def clean_user_activity(df):
    # Drop null records and cast types
    df = df.filter(pl.col("user_id").is_not_null())
    
    # Apply business logic: Sessionize events
    df = df.with_columns([
        pl.col("timestamp").str.to_datetime(),
        (pl.col("user_id") != pl.col("user_id").shift()).alias("new_session")
    ])
    
    return df.select(["user_id", "timestamp", "event_type"])
